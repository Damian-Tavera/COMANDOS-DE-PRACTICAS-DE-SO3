PRACTICA 1

# 1. Actualizar e instalar el paquete del servidor NFS
sudo apt update
sudo apt install nfs-kernel-server -y

# 2. Iniciar y habilitar el servicio
sudo systemctl start nfs-server
sudo systemctl enable nfs-server

# 3. Crear el directorio a compartir y 100 archivos
sudo mkdir -p /var/nfs/os3
cd /var/nfs/os3
sudo touch Adrian{1..100}.txt

# 4. Asignar propiedad (nobody:nogroup) y permisos
sudo chown nobody:nogroup /var/nfs/os3
sudo chmod 755 /var/nfs/os3

# 5. Configurar el archivo /etc/exports (Autorizando a toda la subred 192.168.100.0/24)
sudo nano /etc/exports
> /var/nfs/os3 192.168.100.0/24(rw,sync,no_subtree_check)

# 6. Exportar los recursos para que los cambios surtan efecto
sudo exportfs -rav

# 7. Configurar Firewall (UFW) para permitir el tráfico NFS desde la subred
sudo ufw allow from 192.168.100.0/24 to any port nfs
sudo ufw reload 




# 1. Actualizar e instalar el paquete cliente NFS
sudo apt update
sudo apt install nfs-common -y

# 2. Crear un punto de montaje local
sudo mkdir -p /mnt/os3

# 3. Montar manualmente el recurso NFS y verificar
# La IP del servidor es 192.168.100.249
sudo mount -t nfs 192.168.100.249:/var/nfs/os3 /mnt/os3
df -h
ls /mnt/os3

# 4. Hacer el montaje permanente editando /etc/fstab
sudo nano /etc/fstab
# La línea posterior al signo ">"
> 192.168.100.249:/var/nfs/os3 /mnt/os3 nfs defaults 0 0

# 5. Reiniciar la máquina y verificar la persistencia del montaje
sudo reboot
# Tras el reinicio, verificar el montaje automático:
df -h
ls /mnt/os3

# 1. Actualizar e instalar SAMBA y herramientas necesarias
sudo apt update
sudo apt install samba samba-client samba-common -y

# 2. Habilitar e iniciar los servicios SMB y NMB
sudo systemctl enable --now smbd nmbd

# 3. Crear la carpeta compartida y archivos de prueba:
sudo mkdir -p /srv/samba/public
cd /srv/samba/public
touch Adrian{1..100}.txt

# 4. Configurar permisos y propiedades:
sudo chmod -R 777 /srv/samba/public
sudo chown -R nobody:nogroup /srv/samba/public

# 5. Configurar el Firewall (UFW) para SAMBA:
# La regla 'allow samba' abre todos los puertos requeridos (137, 138, 139, 445).
sudo ufw allow samba
sudo ufw reload

# 1. Editar el archivo de configuración de SAMBA
sudo nano /etc/samba/smb.conf

# 2. Agregar la siguiente sección al final del archivo:
# --- INICIO DE SECCIÓN ---
[public]
    path = /srv/samba/public
    browsable = yes
    writable = yes
    guest ok = yes
    read only = no
    force user = nobody
# --- FIN DE SECCIÓN ---

# 1. Crear el grupo y el usuario 'damiantavera'
sudo groupadd sambausers
sudo useradd -M -s /usr/sbin/nologin damiantavera
sudo usermod -aG sambausers damiantavera

# 2. Asignar la contraseña de SAMBA (Este usuario se usará en Windows)
sudo smbpasswd -a damiantavera
# (Se te pedirá que introduzcas la nueva contraseña de Samba)

# 3. Reiniciar los servicios de SAMBA:
sudo systemctl restart smbd nmbd

# 1. Instalar y configurar iptables

sudo systemctl stop smbd nmbd # Asegura que los servicios de APT no corran

# 2. Detener UFW (Para usar Iptables)
sudo ufw disable

# 3. Limpiar Iptables (Para empezar de cero)
sudo iptables -F
sudo iptables -X

# 4. Establecer las reglas críticas del Controlador de Dominio (CORRECCIÓN DE IPS)
# Asumiendo que el tráfico viene de la subred 192.168.100.0/24
sudo iptables -A INPUT -p tcp -s 192.168.100.0/24 -m state --state NEW -m multiport --dports 53,88,135,139,389,445,464,636,3268,49152:65535 -j ACCEPT
sudo iptables -A INPUT -p udp -s 192.168.100.0/24 -m state --state NEW -m multiport --dports 53,123,137,138,389,636 -j ACCEPT

# 5. Aceptar el tráfico ya establecido
sudo iptables -A INPUT -m state --state RELATED,ESTABLISHED -j ACCEPT

# 6. Guardar las reglas para que persistan en el reinicio
sudo iptables-save | sudo tee /etc/iptables/rules.v4
sudo systemctl restart netfilter-persistent

# 7. Verificar el estado
sudo iptables -L -n
