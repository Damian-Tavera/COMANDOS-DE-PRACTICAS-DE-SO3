sudo apt update

# =================================================================

sudo ufw allow 10000/tcp

# Recargar o habilitar el firewall UFW si no estaba activo
sudo ufw enable

#Crear token de API en DigitalOcean
> API >Generate New Token > Name (terraform-token) >Read and Write' (full access)>Generate Token.

#Crear o subir la llave pública SSH
>Settings >Security >Add SSH Key.

#En nuestro servidor parrot, ejecutamos:
   cat /root/.ssh/id_ed25519.pub


#Copiamos la salida del comando anterior (Si no existe, créala con ssh-keygen) y pégala en DigitalOcean.
sudo apt update
sudo apt install wget software-properties-common -y
wget -O- https://apt.releases.hashicorp.com/gpg | \
    gpg --dearmor | \
    sudo tee /usr/share/keyrings/hashicorp-archive-keyring.gpg > /dev/null
echo "deb [signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg] https://apt.releases.hashicorp.com $(lsb_release -cs) main" | \
    sudo tee /etc/apt/sources.list.d/hashicorp.list
sudo apt update
sudo apt install terraform -y
terraform --version


#Crear carpeta y archivo de configuración Terraform

mkdir ~/terraform-digitalocean
cd ~/terraform-digitalocean

nano main.tf

   terraform {
     required_providers {
       digitalocean = {
         source  = "digitalocean/digitalocean"
         version = "~> 2.0"
       }
     }
   }
   provider "digitalocean" {
     token = var.do_token
   }
   variable "do_token" {}


#Crear archivo de variables para el token
nano terraform.tfvars

#Escribe tu token:
do_token = "TU_TOKEN_DE_DIGITALOCEAN"

#Inicializar Terraform
terraform init

#Crear la configuración del Droplet
nano main.tf

#Añade al final:
   resource "digitalocean_droplet" "os3vm" {
     image    = "ubuntu-22-04-x64"
     name     = "OS3vm"
     region   = "nyc1"
     size     = "s-1vcpu-1gb"
     ssh_keys = [ "TU_FINGERPRINT_DE_LA_LLAVE" ]
   }


#Validar configuración
terraform validate

#Desplegar la máquina virtual
terraform apply

#Validar la VM en el portal de DigitalOcean.

#Por SSH (Reemplaza DIRECCION_IP con la IP pública del droplet)
   ssh root@DIRECCION_IP

Practica 3

# 1. Instalar Ansible
sudo apt update
sudo apt install ansible -y

# 2. Instalar pip3 (si no está instalado)
sudo apt install python3-pip -y

# 3. Instalar dependencias de Python para conectar con Windows (WinRM)
sudo apt install python3-winrm -y
pip3 install pywinrm --break-system-packages
sudo apt autoremove

# 4. Verificar la instalación de Ansible
ansible --version


# 1. Crear el usuario 'ansible' en ambas máquinas Linux (VM-Server y VM-Cliente)
sudo useradd -m ansible
sudo passwd ansible

# 2. Agregar el usuario 'ansible' al grupo 'wheel' o 'sudo' para permisos de administrador
# En Debian/Parrot, el grupo de sudo es 'sudo', no 'wheel'.
sudo usermod -aG sudo ansible


# 3. Permitir sudo sin contraseña (Opcional, pero recomendado para IaC)
echo 'ansible ALL=(ALL) NOPASSWD:ALL' | sudo tee /etc/sudoers.d/ansible

# 4. Crear la clave SSH (Si no existe) en el VM-Server (donde ejecutarás Ansible)
ssh-keygen

# 5. Mostrar la clave pública para copiarla o usar ssh-copy-id
cat ~/.ssh/id_rsa.pub

# 6. Copiar la clave pública al cliente Linux (reemplaza [IP_DEL_CLIENTE_LINUX])
ssh-copy-id ansible@[IP_DEL_CLIENTE_LINUX]

winrm quickconfig
winrm set winrm/config/service/auth '@{Basic="true"}'
winrm set winrm/config/service '@{AllowUnencrypted="true"}'
winrm set winrm/config/listener?Address=*+Transport=HTTP '@{Port="5985"}'

Set-Item -Path WSMan:\localhost\Service\AllowUnencrypted -Value $true
Set-Item -Path WSMan:\localhost\Service\Auth\Basic -Value $true

# Editar el archivo de inventario en Parrot OS
sudo nano /etc/ansible/hosts

[linux]
cliente1 ansible_host=[IP_DEL_CLIENTE_LINUX] ansible_user=ansible ansible_ssh_private_key_file=~/.ssh/id_rsa ansible_become=true ansible_become_method=sudo

[win]
cliente2 ansible_host=[IP_DEL_CLIENTE_WINDOWS] ansible_user=ansible ansible_password=[TU_CONTRASEÑA] ansible_connection=winrm ansible_winrm_transport=basic ansible_port=5985 ansible_winrm_server_cert_validation=ignore

# Probar conectividad con el cliente Linux
ansible linux -m ping

# Probar conectividad con el cliente Windows
ansible win -m win_ping
